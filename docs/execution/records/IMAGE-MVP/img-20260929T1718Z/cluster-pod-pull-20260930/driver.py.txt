def save():
    (EVIDENCE/'result.json').write_text(json.dumps(state,indent=2)+'\n')
def check(name,ok,**facts):
    state['checks'].append(dict(name=name,pass_=bool(ok),**facts));save()
    print(name, 'pass' if ok else 'FAIL',flush=True)
    if not ok: raise RuntimeError(name)
def kube(method,path,body=None,allowed=(200,)):
    if method!='GET':
        assert (path=='/api/v1/namespaces' and method=='POST' and body['metadata']['name'] in NAMESPACES.values()) or any(path.startswith('/api/v1/namespaces/'+ns) for ns in NAMESPACES.values())
    req=urllib.request.Request(API+path,data=None if body is None else json.dumps(body).encode(),method=method,headers={'Content-Type':'application/json'})
    try:
        with urllib.request.urlopen(req,timeout=20) as r: code,raw=r.status,r.read()
    except urllib.error.HTTPError as e: code,raw=e.code,e.read()
    assert code in allowed, f'Kubernetes {method} {path}: {code}'
    return code,json.loads(raw) if raw else None
def harbor(method,path,body=None,allowed=(200,),anonymous=False):
    assert path.startswith('/api/v2.0/')
    cmd=['curl','--silent','--show-error','--max-time','30','--cacert',str(CA),'--request',method,'--dump-header',str(PRIVATE/'response-headers'),'--write-out','\n%{http_code}']
    if not anonymous: cmd+=['--header','@'+str(PRIVATE/'admin-header')]
    if body is not None:cmd+=['--header','Content-Type: application/json','--data-binary','@-']
    p=subprocess.run(cmd+[HARBOR+path],input=None if body is None else json.dumps(body).encode(),stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    assert p.returncode==0, f'Harbor TLS/transport curl exit {p.returncode}'
    raw,code=p.stdout.rsplit(b'\n',1);code=int(code)
    assert code in allowed,f'Harbor {method} {path}: {code}'
    return code,json.loads(raw) if raw else None
def authfile(label,robot):
    auth=base64.b64encode((robot['name']+':'+robot['secret']).encode()).decode()
    value={'auths':{REGISTRY:{'auth':auth}}}
    path=PRIVATE/(label+'-auth.json');path.write_text(json.dumps(value))
    return path,value
def robot(label,projects,actions):
    want={'name':RUN+'-'+label,'description':'ani-image:v1:technical-pod:'+RUN+':'+label,'level':'system','duration':1,'permissions':[{'kind':'project','namespace':p,'access':[{'resource':'repository','action':a,'effect':'allow'} for a in actions]} for p in projects]}
    _,r=harbor('POST','/api/v2.0/robots',want,(201,))
    owned={'id':r['id'],'name':r['name'],'description':want['description'],'permissions':want['permissions']}
    state['robots'].append(owned);save()
    _,actual=harbor('GET','/api/v2.0/robots/'+str(r['id']))
    assert actual['permissions']==want['permissions'] and actual['description']==want['description']
    assert not actual.get('cover_all',False) and actual['expires_at']>time.time()
    return authfile(label,r)
def skopeo(label,args,expect_success=True):
    p=subprocess.run(['skopeo']+args,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=120)
    # Skopeo receives credentials exclusively through private authfiles.
    log=(p.stdout+p.stderr).decode(errors='replace')
    (EVIDENCE/(label+'.txt')).write_text(log)
    if expect_success: assert p.returncode==0, f'{label}: exit {p.returncode}'
    else:check(label,p.returncode!=0 and bool(re.search(r'denied|unauthorized|authentication required',log,re.I)),exit_code=p.returncode)
    return p.stdout
def create_kube(ns,plural,obj):
    obj.setdefault('metadata',{}).setdefault('labels',{})['ani-image-test-run']=RUN
    path='/api/v1/namespaces'+('/'+ns+'/'+plural if ns else '')

    rec={'namespace':ns,'plural':plural,'name':created['metadata']['name'],'uid':created['metadata']['uid']}

def pod(ns,name,node,ref):
    return create_kube(ns,'pods',{'apiVersion':'v1','kind':'Pod','metadata':{'name':name},'spec':{'nodeSelector':{'kubernetes.io/hostname':node},'automountServiceAccountToken':False,'restartPolicy':'Never','activeDeadlineSeconds':900,'terminationGracePeriodSeconds':1,'imagePullSecrets':[{'name':'image-pull'}],'securityContext':{'runAsNonRoot':True,'runAsUser':65534,'seccompProfile':{'type':'RuntimeDefault'}},'containers':[{'name':'check','image':ref,'imagePullPolicy':'Always','command':['/bin/sh','-c','printf "ANI-IMAGE-PULL-OK\\n"; sleep 600'],'securityContext':{'allowPrivilegeEscalation':False,'readOnlyRootFilesystem':True,'capabilities':{'drop':['ALL']}},'resources':{'requests':{'cpu':'10m','memory':'16Mi'},'limits':{'cpu':'100m','memory':'64Mi'}}}]}})
def observe(rec):
    _,p=kube('GET','/api/v1/namespaces/'+rec['namespace']+'/pods/'+rec['name'])
    assert p['metadata']['uid']==rec['uid']
    return p
def wait_positive(records):
    deadline=time.monotonic()+150
    while time.monotonic()<deadline:
        pods=[observe(r) for r in records]
        if all(any(c.get('ready') and c.get('state',{}).get('running') for c in p.get('status',{}).get('containerStatuses',[])) for p in pods):
            for p in pods:
                check('pull-'+p['metadata']['name'],True,uid=p['metadata']['uid'],node=p['spec']['nodeName'],image=p['spec']['containers'][0]['image'],image_pull_policy=p['spec']['containers'][0]['imagePullPolicy'],image_id=p['status']['containerStatuses'][0]['imageID'])
            return
        time.sleep(3)
    for r in records:(EVIDENCE/(r['name']+'-pod.json')).write_text(json.dumps(observe(r),indent=2))
    raise RuntimeError('positive Pod timeout')
def wait_negative(records):
    deadline=time.monotonic()+120
    while time.monotonic()<deadline:
        pods=[observe(r) for r in records]
        statuses=[p.get('status',{}).get('containerStatuses',[{}])[0] for p in pods]
        if any(c.get('containerID') or c.get('state',{}).get('running') for c in statuses):raise RuntimeError('cross-tenant Pod unexpectedly started')
        if all(c.get('state',{}).get('waiting',{}).get('reason') in ('ErrImagePull','ImagePullBackOff') and re.search(r'401|403|denied|unauthorized|authorization failed',c.get('state',{}).get('waiting',{}).get('message',''),re.I) for c in statuses):
            for p,c in zip(pods,statuses):check('cached-cross-tenant-'+p['spec']['nodeName'],True,uid=p['metadata']['uid'],node=p['spec']['nodeName'],waiting=c['state']['waiting'],image=p['spec']['containers'][0]['image'])
            return
        time.sleep(2)
    raise RuntimeError('negative Pod did not show registry authorization rejection')
def cleanup():
    errors=[]
    for rec in reversed(state['kubernetes']):
        try:
            path='/api/v1/namespaces/'+rec['name'] if not rec['namespace'] else '/api/v1/namespaces/'+rec['namespace']+'/'+rec['plural']+'/'+rec['name']
            code,obj=kube('GET',path,allowed=(200,404))
            if code==404:continue
            assert obj['metadata']['uid']==rec['uid'] and obj['metadata']['labels']['ani-image-test-run']==RUN
            if not rec['namespace']:
                for plural in ('pods','secrets','configmaps','services','persistentvolumeclaims','resourcequotas','limitranges'):
                    _,items=kube('GET',path+'/'+plural)
                    for item in items['items']:
                        if plural=='configmaps' and item['metadata']['name']=='kube-root-ca.crt':continue
                        assert any(x['uid']==item['metadata']['uid'] for x in state['kubernetes']), 'unknown namespace member'
            kube('DELETE',path,{'apiVersion':'v1','kind':'DeleteOptions','preconditions':{'uid':rec['uid']},'propagationPolicy':'Foreground'},(200,202))
            state['cleanup'].append({'kind':rec['plural'],'uid':rec['uid'],'delete':'accepted'});save()
        except Exception as e:errors.append('Kubernetes '+rec['name']+': '+str(e))
    for r in reversed(state['robots']):
        try:
            _,actual=harbor('GET','/api/v2.0/robots/'+str(r['id']))
            assert actual['id']==r['id'] and actual['name']==r['name'] and actual['description']==r['description']
            harbor('DELETE','/api/v2.0/robots/'+str(r['id']),allowed=(200,))
            harbor('GET','/api/v2.0/robots/'+str(r['id']),allowed=(404,))
            state['cleanup'].append({'kind':'robot','id':r['id'],'absent':True});save()
        except Exception as e:errors.append('robot '+str(r['id'])+': '+str(e))
    for r in reversed(state['projects']):
        try:
            _,p=harbor('GET','/api/v2.0/projects/'+str(r['id']))
            assert p['project_id']==r['id'] and p['name']==r['name'] and p['creation_time']==r['creation_time']
            _,repos=harbor('GET','/api/v2.0/projects/'+str(r['id'])+'/repositories?page_size=100')
            assert all(x['name']==r['name']+'/fixture' for x in repos),'unknown project repository'
            for repo in repos:
                _,artifacts=harbor('GET','/api/v2.0/projects/'+str(r['id'])+'/repositories/fixture/artifacts?page_size=100')
                assert all(a['digest']==state.get('image_digest') for a in artifacts),'unknown project artifact'
                for a in artifacts:harbor('DELETE','/api/v2.0/projects/'+str(r['id'])+'/repositories/fixture/artifacts/'+a['digest'],allowed=(200,))
                harbor('DELETE','/api/v2.0/projects/'+str(r['id'])+'/repositories/fixture',allowed=(200,))
            harbor('DELETE','/api/v2.0/projects/'+str(r['id']),allowed=(200,))
            harbor('GET','/api/v2.0/projects/'+str(r['id']),allowed=(404,))
            state['cleanup'].append({'kind':'project','id':r['id'],'absent':True});save()
        except Exception as e:errors.append('project '+str(r['id'])+': '+str(e))
    deadline=time.monotonic()+60
    while time.monotonic()<deadline:
        remaining=[r for r in state['kubernetes'] if not r['namespace'] and kube('GET','/api/v1/namespaces/'+r['name'],allowed=(200,404))[0]!=404]
        if not remaining:break
        time.sleep(2)
    else:errors.append('namespace deletion not finished')
    state['cleanup_errors']=errors;save()
    if not errors:
        for path in PRIVATE.iterdir():
            if path.is_file():path.unlink()
    return not errors
