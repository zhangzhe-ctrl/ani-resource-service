import base64, hashlib, json, os, subprocess, urllib.request
from pathlib import Path
os.umask(0o077)
run = Path.home() / 'ani-image-mvp-runs/pod-pull-20260930T0200Z'
run.mkdir(mode=0o700, exist_ok=False)
(run/'private').mkdir(mode=0o700)
(run/'evidence').mkdir()
base='http://127.0.0.1:46483'
def get(path):
    return json.load(urllib.request.urlopen(base+path, timeout=15))
assert get('/api/v1/namespaces/kube-system')['metadata']['uid']=='87ecef8e-ac4e-442b-8e15-5e906263be6b'
ca=Path.home()/'ani-image-mvp-runs/cluster-assessment-20260930T0110Z/image-harbor-ca-public.pem'
assert hashlib.sha256(ca.read_bytes()).hexdigest()=='59e87314e44f89c3c562bdaa0535da00b45567711bd013a1ae268299406e7b28'
data=get('/api/v1/namespaces/ani-harbor/secrets/ani-harbor-admin')['data']
password=base64.b64decode(data['HARBOR_ADMIN_PASSWORD'])
header=run/'private/admin-header'
header.write_bytes(b'Authorization: Basic '+base64.b64encode(b'admin:'+password)+b'\n')
del password,data
cmd=['curl','--silent','--show-error','--fail','--max-time','20','--cacert',str(ca),'--header','@'+str(header),'https://172.16.101.10:30003/api/v2.0/systeminfo']
p=subprocess.run(cmd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
if p.returncode: raise RuntimeError('authenticated Harbor request failed, curl exit '+str(p.returncode))
d=json.loads(p.stdout)
info={k:d.get(k) for k in ['harbor_version','registry_url','auth_mode']}
(run/'evidence/harbor-info.json').write_text(json.dumps(info,indent=2)+'\n')
print(json.dumps(info),flush=True)
p=subprocess.run(['skopeo','inspect','--override-os','linux','--override-arch','amd64','docker://docker.io/library/busybox:1.37.0'],stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=90)
if p.returncode:
    (run/'evidence/base-inspect-error.txt').write_bytes(p.stderr)
    raise RuntimeError('base image inspect failed; see bounded stderr evidence')
d=json.loads(p.stdout)
base_image={k:d.get(k) for k in ['Name','Digest','Architecture','Os','Layers']}
(run/'evidence/base-image.json').write_text(json.dumps(base_image,indent=2)+'\n')
print(json.dumps(base_image))
